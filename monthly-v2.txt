function getItems(row, targetColumns){
    tvalue = targetColumns.map(v => ws[`${v}${row}`]?.v ?? null);
    return {
        "budget": tvalue[0],
        "forecast": tvalue[1],
        "actual": tvalue[2],
        "variance": tvalue[3],
        "remarks": tvalue[4]
    };
};

function getCost(category, subCategory, item, row){
    return {    
        'category': category,
        'subCategory': subCategory,
        'item': item,
        'data': {
            'monthly': getItems(row, ['H', 'L', 'M', 'N', 'O']),
            'ytd': getItems(row, ['Q', 'U', 'V', 'W', 'Y'] ),
            'outlook': getItems(row, ['AA', 'AE', 'AF', 'AG', 'AH']),
        }
    }
}

const dataset = Object.entries(cost_item_names).flatMap(([cat, v1]) =>
    Object.entries(v1).flatMap(([scat, v2]) =>
        Object.entries(v2).flatMap(([itm, r]) =>
            getCost(cat, scat, itm, r)
        )
    )
);

const ymlDataset = yaml.dump(dataset);
